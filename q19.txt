Question 3 Template 2 - Dynamic Table
Q3. Design a web page to accept Student Name, Register Number and Department. Write JavaScript code to dynamically add
the student details as a new row in an HTML table. Provide an option to delete a selected row.
index.html
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>Student Table</title>
<style>
body{font-family:Arial,sans-serif;background:#eaf2f8;text-align:center;margin:30px}
.box{background:#fff;max-width:460px;margin:auto;padding:20px;border-radius:10px;box-shadow:0 2px 8px rgba(0,0,0,.25)}
input,textarea,select{padding:8px;margin:6px;border:1px solid #888;border-radius:5px;width:85%}
button{background:#2874a6;color:#fff;border:0;padding:9px 18px;margin:5px;border-radius:5px;cursor:pointer}
.err{color:red}.ok{color:green}
table{width:100%;border-collapse:collapse;margin-top:10px}
th,td{border:1px solid #999;padding:6px}th{background:#2874a6;color:#fff}
</style>
</head>
<body>
<div class="box"><h2>Student Details</h2>
<input id="name" placeholder="Student Name">
<input id="reg" placeholder="Register Number">
<input id="dept" placeholder="Department"><br>
<button onclick="addRow()">Add</button>
<button onclick="delRow()" style="background:#c0392b">Delete Selected</button>
<table id="tbl"><tr><th>Select</th><th>Name</th><th>Reg No</th><th>Department</th></tr></table></div>
<script>
function addRow() {
  let n = document.getElementById("name").value;
  let r = document.getElementById("reg").value;
  let d = document.getElementById("dept").value;
  if (n == "" || r == "" || d == "") { alert("Fill all fields"); return; }
  let row = document.getElementById("tbl").insertRow();
  row.innerHTML = "<td><input type='checkbox'></td><td>" + n + "</td><td>" + r + "</td><td>" + d + "</td>";
  document.getElementById("name").value = "";
  document.getElementById("reg").value = "";
  document.getElementById("dept").value = "";
}
function delRow() {
  document.querySelectorAll("#tbl input[type=checkbox]").forEach(c => {
    if (c.checked) c.parentElement.parentElement.remove();
  });
}
</script>
</body>
</html>
