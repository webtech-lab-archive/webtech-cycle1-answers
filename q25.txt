<script>
function checkEmail() {
  let e = document.getElementById("email").value;
  let pattern = /^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$/;
  let m = document.getElementById("msg");
  if (pattern.test(e)) { m.className = "ok"; m.innerText = e + " is a VALID email address"; }
  else { m.className = "err"; m.innerText = e + " is an INVALID email address"; }
}
</script>
