Question 5 Template 1 - DOM Button Events
Q5. Design a webpage containing a textarea. Write JavaScript code to dynamically display the number of characters entered
by the user whenever the content changes.
index.html
<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>Character Counter</title>
<style>
body{font-family:Arial,sans-serif;background:#eaf2f8;text-align:center;margin:30px}
.box{background:#fff;max-width:460px;margin:auto;padding:20px;border-radius:10px;box-shadow:0 2px 8px rgba(0,0,0,.25)}
input,textarea,select{padding:8px;margin:6px;border:1px solid #888;border-radius:5px;width:85%}
button{background:#2874a6;color:#fff;border:0;padding:9px 18px;margin:5px;border-radius:5px;cursor:pointer}
.err{color:red}.ok{color:green}
table{width:100%;border-collapse:collapse;margin-top:10px}
th,td{border:1px solid #999;padding:6px}th{background:#2874a6;color:#fff}
</style>
</head>
<body>
<div class="box"><h2>Character Counter</h2>
<textarea id="txt" rows="5" placeholder="Type here..." oninput="count()"></textarea>
<p>Characters entered: <b id="cnt">0</b></p></div>
<script>
function count() {
  let text = document.getElementById("txt").value;
  document.getElementById("cnt").innerText = text.length;
}
</script>
</body>
</html>
