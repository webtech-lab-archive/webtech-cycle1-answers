Q8. Design a password validation form. Write a JavaScript program to check whether the password contains at least 8
characters, one uppercase letter, one lowercase letter, one digit and one special character.
index.html

<style>

<div class="box"><h2>Password Validation</h2>
<input type="password" id="pw" placeholder="Enter password"><br>
<button onclick="check()">Check Password</button>
<ul id="list" style="list-style:none;padding:0;text-align:left;margin-left:60px"></ul>
<p id="msg"></p></div>
<script>
function check() {
  let p = document.getElementById("pw").value;
  let rules = [
    [p.length >= 8, "At least 8 characters"],
    [/[A-Z]/.test(p), "One uppercase letter"],
    [/[a-z]/.test(p), "One lowercase letter"],
    [/\d/.test(p), "One digit"],
    [/[^A-Za-z0-9]/.test(p), "One special character"]
  ];
  let out = "", all = true;
  rules.forEach(r => {
    out += "<li class='" + (r[0] ? "ok" : "err") + "'>" + (r[0] ? "&#10004; " : "&#10008; ") + r[1] + "</li>";
    if (!r[0]) all = false;
  });
  document.getElementById("list").innerHTML = out;
  document.getElementById("msg").innerHTML = all ? "<b class='ok'>Strong password</b>" : "<b class='err'>Weak 
password</b>";
}
</script>
