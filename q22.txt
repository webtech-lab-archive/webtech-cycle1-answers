<script>
function calcAge() {
  let v = document.getElementById("dob").value;
  let m = document.getElementById("msg");
  if (v == "") { m.className = "err"; m.innerText = "Please enter your date of birth"; return; }
  let dob = new Date(v), today = new Date();
  if (dob > today) { m.className = "err"; m.innerText = "Invalid date: future date not allowed"; return; }
  let age = today.getFullYear() - dob.getFullYear();
  let diff = today.getMonth() - dob.getMonth();
  if (diff < 0 || (diff == 0 && today.getDate() < dob.getDate())) age--;
  m.className = "ok";
  m.innerText = "Valid date. Your age is " + age + " years";
}
</script>
