Q7. Develop a registration form with Password and Confirm Password fields. Write a JavaScript program to verify whether both
passwords match. Display suitable messages based on the validation result.

<style>

<div class="box"><h2>Registration Form</h2>
<input type="text" id="user" placeholder="Username"><br>
<input type="password" id="pw" placeholder="Password"><br>
<input type="password" id="cpw" placeholder="Confirm Password"><br>
<button onclick="verify()">Register</button>
<p id="msg"></p></div>
<script>
function verify() {
  let p = document.getElementById("pw").value;
  let c = document.getElementById("cpw").value;
  let m = document.getElementById("msg");
  if (p == "" || c == "") { m.className = "err"; m.innerText = "Password fields cannot be empty"; }
  else if (p == c) { m.className = "ok"; m.innerText = "Passwords match. Registration successful!"; }
  else { m.className = "err"; m.innerText = "Passwords do not match. Try again."; }
}
</script>
