<script>
function check(id, re, msg) {
  let ok = re.test(document.getElementById(id).value);
  document.getElementById(id + "Err").innerText = ok ? "" : msg;
  return ok;
}
function validate() {
  let r = [
    check("name", /^[A-Za-z ]{3,}$/, "Name must have 3+ letters"),
    check("reg", /^\d{2}[A-Z]{2,3}\d{3}$/, "Format should be like 21IT014"),
    check("email", /^[^\s@]+@[^\s@]+\.[a-z]{2,}$/i, "Enter a valid email"),
    check("mobile", /^[6-9]\d{9}$/, "10 digits starting with 6-9"),
    check("pass", /^(?=.*[A-Za-z])(?=.*\d).{8,}$/, "Min 8 chars with letter and digit")
  ];
  document.getElementById("msg").innerHTML = r.every(x => x) ? "<b class='ok'>Registration successful!</b>" : "";
}
</script>
