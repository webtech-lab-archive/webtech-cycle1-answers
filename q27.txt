<script>
function loadData() {
  let xhr = new XMLHttpRequest();
  xhr.onreadystatechange = function () {
    if (xhr.readyState == 4 && xhr.status == 200)
      document.getElementById("result").innerText = xhr.responseText;
  };
  xhr.open("GET", "students.txt", true);
  xhr.send();
}
</script>
