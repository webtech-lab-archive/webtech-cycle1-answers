function addRow() {
  let n = document.getElementById("name").value;
  let r = document.getElementById("reg").value;
  let d = document.getElementById("dept").value;

  document.getElementById("reg").value = "";
  document.getElementById("dept").value = "";
}
function delRow() {
  document.querySelectorAll("#tbl input[type=checkbox]").forEach(c => {
    if (c.checked) c.parentElement.parentElement.remove();
  });
}

function calcAge() {
  let v = document.getElementById("dob").value;
  let m = document.getElementById("msg");
  if (v == "") { m.className = "err"; m.innerText = "Please enter your date of birth"; return; }
  let dob = new Date(v), today = new Date();
  if (dob > today) { m.className = "err"; m.innerText = "Invalid date: future date not allowed"; return; }
  let age = today.getFullYear() - dob.getFullYear();
  let diff = today.getMonth() - dob.getMonth();
  if (diff < 0 || (diff == 0 && today.getDate() < dob.getDate())) age--;
  m.className = "ok";
  m.innerText = "Valid date. Your age is " + age + " years";
}
</script>
</body>
</html>

function check() {
  let p = document.getElementById("pw").value;
  let rules = [
    [p.length >= 8, "At least 8 characters"],
    [/[A-Z]/.test(p), "One uppercase letter"],
    [/[a-z]/.test(p), "One lowercase letter"],
    [/\d/.test(p), "One digit"],
    [/[^A-Za-z0-9]/.test(p), "One special character"]
  ];
  let out = "", all = true;
  rules.forEach(r => {
    out += "<li class='" + (r[0] ? "ok" : "err") + "'>" + (r[0] ? "&#10004; " : "&#10008; ") + r[1] + "</li>";
    if (!r[0]) all = false;
  });
  document.getElementById("list").innerHTML = out;
  document.getElementById("msg").innerHTML = all ? "<b class='ok'>Strong password</b>" : "<b class='err'>Weak 
password</b>";
}
</script>
</body>

in an HTML table.
Run: npm install mysql2 | node server.js | open http://localhost:3000
setup.sql (run once in MySQL)
CREATE DATABASE college;
USE college;
CREATE TABLE students(id INT PRIMARY KEY, name VARCHAR(50), regno VARCHAR(15), dept VARCHAR(20));
INSERT INTO students VALUES (1,'Arun Kumar','21IT001','IT'),(2,'Priya Devi','21IT014','IT'),(3,'Karthik 
Raj','21CS007','CSE');
server.js
const http = require('http');
const mysql = require('mysql2');
const db = mysql.createConnection({
  host: 'localhost', user: 'root', password: 'root', database: 'college'
});
db.connect(err => {
  if (err) throw err;
  console.log('MySQL connected');
});
http.createServer((req, res) => {
  db.query('SELECT * FROM students', (err, rows) => {
    if (err) { res.end('Database error'); return; }
    let html = '<h2>Student Records</h2><table border="1" cellpadding="8" style="border-collapse:collapse">' +
               '<tr><th>ID</th><th>Name</th><th>Reg No</th><th>Department</th></tr>';
    rows.forEach(r => {
      html += `<tr><td>${r.id}</td><td>${r.name}</td><td>${r.regno}</td><td>${r.dept}</td></tr>`;
    });
    res.writeHead(200, { 'Content-Type': 'text/html' });
    res.end(html + '</table>');
  });
}).listen(3000, () => console.log('Server running at http://localhost:3000'));

  if (u.pathname == '/calc') {
    const a = parseFloat(u.searchParams.get('a'));
    const b = parseFloat(u.searchParams.get('b'));
    html += `<h3>Results</h3>
      <p>Addition: ${a + b}</p>
      <p>Subtraction: ${a - b}</p>
      <p>Multiplication: ${a * b}</p>
      <p>Division: ${b == 0 ? 'Cannot divide by zero' : a / b}</p>`;
  }
  res.writeHead(200, { 'Content-Type': 'text/html' });
  res.end(html);
}).listen(3000, () => console.log('Server running at http://localhost:3000'));

Q15. Develop a Node.js application to display the current date and time on a webpage using the built-in Date object.
Run: node server.js | open http://localhost:3000
server.js
const http = require('http');
http.createServer((req, res) => {
  const now = new Date();
  res.writeHead(200, { 'Content-Type': 'text/html' });
  res.end(`<meta http-equiv="refresh" content="1">
    <body style="text-align:center;font-family:Arial;margin-top:60px">
    <h1>Current Date and Time</h1>
    <h2>Date: ${now.toDateString()}</h2>
    <h2>Time: ${now.toLocaleTimeString()}</h2>
    <p>${now}</p></body>`);
}).listen(3000, () => console.log('Server running at http://localhost:3000'));
