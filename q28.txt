<script>
function search() {
  let key = document.getElementById("q").value.toLowerCase();
  let xhr = new XMLHttpRequest();
  xhr.onload = function () {
    let books = JSON.parse(xhr.responseText);
    let rows = "<table><tr><th>ID</th><th>Title</th><th>Author</th><th>Availability</th></tr>";
    books.filter(b => b.title.toLowerCase().includes(key) || b.author.toLowerCase().includes(key))
         .forEach(b => rows += "<tr><td>" + b.id + "</td><td>" + b.title + "</td><td>" + b.author +
                    "</td><td>" + (b.available ? "Available" : "Issued") + "</td></tr>");
    document.getElementById("result").innerHTML = rows + "</table>";
  };
  xhr.open("GET", "books.json", true);
  xhr.send();
}
</script>
